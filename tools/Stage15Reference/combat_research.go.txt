package main

var researchCase = os.Getenv("RESEARCH_CASE")
var researchFrame = -1
var researchBaseTick = -1
var researchIntroStart = -1
var researchIntroFrame = -1
var researchModeFile *os.File
var researchModeEncoder *json.Encoder
var researchMode2 = strings.HasPrefix(researchCase, "mode2-")
var researchScenario = strings.TrimPrefix(strings.TrimPrefix(researchCase, "mode2-"), "default-")
var researchStage17 = strings.HasPrefix(researchScenario, "damage-")
var researchDamageScenario = strings.TrimPrefix(researchScenario, "damage-")
var researchDamageMirror = researchStage17 && strings.HasPrefix(researchDamageScenario, "mirror-")
var researchStage18 = strings.HasPrefix(researchScenario, "reentry-")
var researchReentryScenario = strings.TrimPrefix(researchScenario, "reentry-")
var researchReentryMirror = researchStage18 && strings.HasPrefix(researchReentryScenario, "mirror-")
var researchStage19 = strings.HasPrefix(researchScenario, "special-")
var researchSpecialScenario = strings.TrimPrefix(researchScenario, "special-")
var researchStage20 = strings.HasPrefix(researchScenario, "near-wave-")
var researchStage21 = strings.HasPrefix(researchScenario, "heavy-wave-")
var researchWaveTrace = researchStage19 || researchStage20 || researchStage21
var researchSpecialMirror = researchStage19 && strings.HasPrefix(researchSpecialScenario, "mirror-") ||
	researchStage20 && strings.HasPrefix(strings.TrimPrefix(researchScenario, "near-wave-"), "mirror-") ||
	researchStage21 && strings.HasPrefix(strings.TrimPrefix(researchScenario, "heavy-wave-"), "mirror-")
const researchPowerWaveCommand = "\x8e\xe3\x83\x70\x83\x8f\x81\x5b\x83\x45\x83\x46\x83\x43\x83\x75"
var researchStage16 = !researchStage21 && (strings.HasPrefix(researchScenario, "light-") || strings.HasPrefix(researchScenario, "heavy-"))
var researchTraceFile *os.File
var researchEncoder *json.Encoder

func researchBegin() {
	if researchCase == "" {
		return
	}
	if researchBaseTick < 0 && sys.roundState() != 2 {
		if sys.roundState() == 1 && researchMode2 {
			if researchIntroStart < 0 {
				researchIntroStart = sys.tickCount
				var err error
				researchModeFile, err = os.Create(os.Getenv("RESEARCH_MODE_TRACE"))
				if err != nil {
					panic(err)
				}
				researchModeEncoder = json.NewEncoder(researchModeFile)
			}
			researchIntroFrame = sys.tickCount - researchIntroStart
		}
		return
	}

	if researchBaseTick < 0 {
		researchBaseTick = sys.tickCount
		researchIntroFrame = -1
		if researchModeFile != nil {
			if err := researchModeFile.Close(); err != nil {
				panic(err)
			}
			researchModeEncoder = nil
		}
		distance, err := strconv.ParseFloat(os.Getenv("RESEARCH_DISTANCE"), 32)
		if err != nil || distance <= 0 {
			panic("RESEARCH_DISTANCE must be positive")
		}
		researchTraceFile, err = os.Create(os.Getenv("RESEARCH_TRACE"))
		if err != nil {
			panic(err)
		}
		researchEncoder = json.NewEncoder(researchTraceFile)
		sys.randseed = 1337
		if researchStage17 || researchStage18 || researchWaveTrace {
			seed, err := strconv.ParseInt(os.Getenv("RESEARCH_SEED"), 10, 32)
			if err != nil || seed <= 0 || seed >= int64(IMax) {
				panic("RESEARCH_SEED must be between 1 and 2147483646")
			}
			sys.randseed = int32(seed)
		}

		for pn := 0; pn < 2; pn++ {
			c := sys.chars[pn][0]
			sign := float32(1 - 2*pn)
			if researchScenario == "mirror" || researchStage16 && strings.HasSuffix(researchScenario, "-mirror") || researchDamageMirror || researchReentryMirror || researchSpecialMirror {
				sign = -sign
			}
			c.pos, c.facing = [3]float32{-sign * float32(distance) / 2, 0, 0}, sign
			c.oldPos, c.interPos, c.vel = c.pos, c.pos, [3]float32{}
		}
	}

	researchFrame = sys.tickCount - researchBaseTick
	researchTrace("before-command")
}

func researchBits(pn int) InputBits {
	if researchFrame < 0 {
		if pn == 0 && researchMode2 && sys.roundState() == 1 {
			switch researchIntroFrame {
			case 1, 3, 9:
				return IB_PU
			case 5, 7, 11:
				return IB_PD
			}
		}
		return 0
	}
	if researchStage16 {
		return researchCloseNormalBits(pn)
	}
	if researchStage17 {
		return researchDamageBits(pn)
	}
	if researchStage18 {
		return researchReentryBits(pn)
	}
	if researchStage19 {
		return researchSpecialBits(pn)
	}
	if researchStage20 {
		return researchNearWaveBits(pn)
	}
	if researchStage21 {
		return researchHeavyWaveBits(pn)
	}
	if pn == 0 && researchFrame == 5 {
		return IB_Y
	}
	if pn == 1 {
		switch researchScenario {
		case "trade":
			if researchFrame == 7 {
				return IB_X
			}
		case "interrupt":
			if researchFrame == 5 {
				return IB_X
			}
		case "guard":
			if researchFrame >= 9 && researchFrame <= 39 {
				return IB_PR
			}
		case "crouch-guard":
			if researchFrame >= 5 && researchFrame <= 39 {
				return IB_PR | IB_PD
			}
		case "crouch-guard-late":
			if researchFrame >= 10 && researchFrame <= 39 {
				return IB_PR | IB_PD
			}
		}
	}
	return 0
}

func researchSpecialBits(pn int) InputBits {
	scenario := strings.TrimPrefix(researchSpecialScenario, "mirror-")
	parts := strings.Split(scenario, "-")
	button, _ := strconv.Atoi(parts[len(parts)-1])
	forward := IB_PR
	if researchSpecialMirror {
		forward = IB_PL
	}
	first := 5
	if strings.HasPrefix(scenario, "near-whiff-") {
		first = 13
	}
	dual := strings.HasPrefix(scenario, "dual-")
	if pn == 1 && dual {
		if researchSpecialMirror {
			forward = IB_PR
		} else {
			forward = IB_PL
		}
	}
	if pn == 0 || dual {
		if researchFrame == first && !strings.HasPrefix(scenario, "neutral-") {
			return IB_X
		}
		if strings.Contains(scenario, "-no-motion-") {
			if researchFrame == button {
				return IB_X
			}
			return 0
		}
		if button-3 == first && researchFrame == first-1 {
			return IB_PD
		}
		switch researchFrame {
		case button - 3:
			return IB_PD
		case button - 2:
			return IB_PD | forward
		case button - 1:
			if strings.HasPrefix(scenario, "terminal-heavy16-") {
				return forward | IB_Y
			}
			return forward
		case button:
			return forward | IB_X
		}
		if strings.Contains(scenario, "-held-") && researchFrame > button && researchFrame <= button+12 {
			return IB_X
		}
		return 0
	}
	if strings.HasPrefix(scenario, "guard-") && researchFrame >= 7 && researchFrame <= 62 {
		return forward
	}
	if strings.HasPrefix(scenario, "near-whiff-") && researchFrame == 0 {
		return IB_PU
	}
	return 0
}

func researchNearWaveBits(pn int) InputBits {
	scenario := strings.TrimPrefix(strings.TrimPrefix(researchScenario, "near-wave-"), "mirror-")
	parts := strings.Split(scenario, "-")
	button, _ := strconv.Atoi(parts[len(parts)-1])
	jump := strings.HasPrefix(scenario, "whiff-")
	dual := strings.HasPrefix(scenario, "dual-")
	mixed := scenario == "mixed-sequential"
	forward := IB_PR
	if researchSpecialMirror != (pn == 1 && (dual || mixed)) {
		forward = IB_PL
	}
	first := 5
	if jump {
		first = 13
	}
	if mixed {
		button = 20
		if pn == 1 {
			if researchFrame >= 7 && researchFrame <= 62 {
				if forward == IB_PR {
					return IB_PL
				}
				return IB_PR
			}
			first, button = 90, 94
		}
	}
	if pn == 0 || dual || mixed {
		if researchFrame == first {
			return IB_X
		}
		if strings.HasPrefix(scenario, "whiff-terminal-heavy-") {
			if researchFrame == button-1 {
				if strings.Contains(scenario, "-wave-") {
					return forward | IB_Y
				}
				return IB_Y
			}
			if strings.Contains(scenario, "-light-") {
				if researchFrame == button {
					return IB_X
				}
				return 0
			}
		}
		if strings.Contains(scenario, "no-motion") || strings.HasPrefix(scenario, "reentry-") {
			if researchFrame == button {
				return IB_X
			}
			return 0
		}
		if button-3 == first && researchFrame == first-1 {
			return IB_PD
		}
		switch researchFrame {
		case button - 3:
			return IB_PD
		case button - 2:
			return IB_PD | forward
		case button - 1:
			return forward
		case button:
			return forward | IB_X
		}
		if strings.Contains(scenario, "-held-") && researchFrame > button && researchFrame <= button+12 {
			return IB_X
		}
		return 0
	}
	if strings.Contains(scenario, "guard-") && researchFrame >= 7 && researchFrame <= 62 {
		if strings.HasPrefix(scenario, "crouch-guard-") {
			return forward | IB_PD
		}
		return forward
	}
	if jump && researchFrame == 0 {
		return IB_PU
	}
	return 0
}

func researchHeavyWaveBits(pn int) InputBits {
	scenario := strings.TrimPrefix(strings.TrimPrefix(researchScenario, "heavy-wave-"), "mirror-")
	parts := strings.Split(scenario, "-")
	button, _ := strconv.Atoi(parts[len(parts)-1])
	forward, back := IB_PR, IB_PL
	if researchSpecialMirror {
		forward, back = back, forward
	}
	whiff := strings.HasPrefix(scenario, "whiff-")
	first := 5
	if whiff {
		first = 13
	}
	if pn == 1 {
		switch {
		case strings.HasPrefix(scenario, "guard-release-"):
			last := 8
			if button == 34 {
				last = 42
			}
			if researchFrame >= 7 && researchFrame <= last {
				return forward
			}
		case strings.HasPrefix(scenario, "second-guard-") && researchFrame >= 5 && researchFrame <= 67:
			return forward
		case strings.HasPrefix(scenario, "second-only-") && researchFrame == 5:
			return forward
		case strings.HasPrefix(scenario, "guard-") && researchFrame >= 7 && researchFrame <= 67:
			return forward
		case strings.HasPrefix(scenario, "crouch-guard-") && researchFrame >= 7 && researchFrame <= 67:
			return forward | IB_PD
		case whiff && researchFrame == 0:
			return IB_PU
		case scenario == "mixed-sequential" && researchFrame >= 7 && researchFrame <= 67:
			return forward
		case scenario == "mixed-sequential" && researchFrame == 94:
			return IB_X
		case scenario == "mixed-sequential" && researchFrame == 95:
			return IB_PD
		case scenario == "mixed-sequential" && researchFrame == 96:
			return IB_PD | back
		case scenario == "mixed-sequential" && researchFrame == 97:
			return back
		case scenario == "mixed-sequential" && researchFrame == 98:
			return back | IB_X
		}
		return 0
	}
	if researchFrame == first {
		return IB_Y
	}
	if scenario == "mixed-sequential" {
		button = 20
	}
	if button <= 0 {
		return 0
	}
	if strings.Contains(scenario, "-light-") {
		if researchFrame == button {
			return IB_X
		}
		return 0
	}
	if button-3 == first && researchFrame == first-1 {
		return IB_PD
	}
	switch researchFrame {
	case button - 3:
		return IB_PD
	case button - 2:
		return IB_PD | forward
	case button - 1:
		return forward
	case button:
		return forward | IB_X
	}
	if strings.Contains(scenario, "-held-") && researchFrame > button && researchFrame <= button+12 {
		return IB_X
	}
	return 0
}

func researchReentryBits(pn int) InputBits {
	scenario := strings.TrimPrefix(researchReentryScenario, "mirror-")
	first := 5
	jump := strings.HasPrefix(scenario, "approach-") || strings.HasPrefix(scenario, "near-jump-")
	if jump {
		first = 13
	}
	parts := strings.Split(scenario, "-")
	second, _ := strconv.Atoi(parts[len(parts)-1])
	if pn == 0 {
		if researchFrame == first || (scenario == "held" && researchFrame >= first && researchFrame <= 40) {
			return IB_X
		}
		if scenario == "whiff-heavy16-light17" {
			switch researchFrame {
			case 16:
				return IB_Y
			case 17:
				return IB_X
			}
			return 0
		}
		if second == 0 || researchFrame < second {
			return 0
		}
		if researchFrame > second && (!strings.Contains(scenario, "-held-") || researchFrame > second+8) {
			return 0
		}
		if strings.Contains(scenario, "-both-") {
			return IB_X | IB_Y
		}
		if strings.Contains(scenario, "-heavy-") {
			return IB_Y
		}
		if strings.Contains(scenario, "-down-") {
			return IB_X | IB_PD
		}
		return IB_X
	}
	back, forward := IB_PR, IB_PL
	if researchReentryMirror {
		back, forward = forward, back
	}
	if jump && researchFrame == 0 {
		if strings.HasPrefix(scenario, "approach-") {
			return IB_PU | forward
		}
		return IB_PU
	}
	if researchFrame >= 7 && researchFrame <= 40 {
		if strings.Contains(scenario, "-guard-") {
			return back
		}
		if strings.Contains(scenario, "-crouch-") {
			return back | IB_PD
		}
	}
	return 0
}

func researchDamageBits(pn int) InputBits {
	scenario := strings.TrimPrefix(researchDamageScenario, "mirror-")
	if pn == 0 {
		attackFrame := 5
		switch scenario {
		case "carry-expired":
			attackFrame = 8
		case "carry-last":
			attackFrame = 6
		}
		if researchFrame == attackFrame {
			if scenario == "light-near" || scenario == "light-far" || scenario == "near-trade" {
				return IB_X
			}
			return IB_Y
		}
		return 0
	}
	switch scenario {
	case "interrupt", "near-heavy-trade":
		if researchFrame == 5 {
			return IB_X
		}
	case "far-trade":
		if researchFrame == 7 {
			return IB_X
		}
	case "near-trade":
		if researchFrame == 5 {
			return IB_Y
		}
	case "second-only":
		if researchFrame == 5 {
			if researchDamageMirror {
				return IB_PL
			}
			return IB_PR
		}
	case "heavy-guard", "heavy-far-guard":
		guardFrame := 7
		if scenario == "heavy-far-guard" {
			guardFrame = 4
		}
		if researchFrame >= guardFrame {
			if researchDamageMirror {
				return IB_PL
			}
			return IB_PR
		}
	case "carry-live":
		if researchFrame == 2 {
			return IB_X
		}
	case "carry-last", "carry-expired", "carry-edge":
		if researchFrame == 1 {
			return IB_X
		}
	}
	return 0
}

func researchCloseNormalBits(pn int) InputBits {
	light := strings.HasPrefix(researchScenario, "light-")
	scenario := strings.TrimPrefix(strings.TrimPrefix(researchScenario, "light-"), "heavy-")
	attackFrame := 5
	if scenario == "whiff" {
		attackFrame = 13
	}
	if pn == 0 && researchFrame == attackFrame {
		if strings.HasPrefix(scenario, "both-") {
			return IB_X | IB_Y
		}
		if light {
			return IB_X
		}
		return IB_Y
	}
	if pn != 1 {
		return 0
	}
	switch scenario {
	case "guard":
		if researchFrame >= 7 && researchFrame <= 55 {
			return IB_PR
		}
	case "crouch-guard":
		if researchFrame >= 5 && researchFrame <= 55 {
			return IB_PR | IB_PD
		}
	case "crouch-late":
		if researchFrame >= 8 && researchFrame <= 55 {
			return IB_PR | IB_PD
		}
	case "guard-release":
		if researchFrame >= 7 && researchFrame <= 8 {
			return IB_PR
		}
	case "crouch-release", "crouch-to-stand":
		if researchFrame >= 5 && researchFrame <= 8 {
			return IB_PR | IB_PD
		}
		if scenario == "crouch-to-stand" && researchFrame >= 9 && researchFrame <= 55 {
			return IB_PR
		}
	case "whiff":
		if researchFrame == 0 {
			return IB_PU
		}
	case "second-only":
		if researchFrame == 5 {
			return IB_PR
		}
	case "second-guard":
		if researchFrame >= 5 && researchFrame <= 55 {
			return IB_PR
		}
	case "trade":
		if researchFrame == 5 {
			if light {
				return IB_Y
			}
			return IB_X
		}
	}
	return 0
}

func researchInput(c *Char, original [14]bool) [14]bool {
	if researchCase == "" || c == nil ||
		(c.helperIndex != 0 && c.helperId != 7600 && !((researchStage18 || researchWaveTrace) && c.helperId == 99999)) {
		return original
	}
	return researchBits(c.playerNo).BitsToKeys()
}

func researchNumbers(numbers ...float32) []interface{} {
	values := make([]interface{}, len(numbers))
	for i, number := range numbers {
		if !math.IsNaN(float64(number)) && !math.IsInf(float64(number), 0) {
			values[i] = number
		}
	}
	return values
}

func researchTrace(phase string) {
	researchTraceWithOwner(phase, 0)
}

func researchTraceWithOwner(phase string, owner int) {
	if researchEncoder == nil {
		if researchModeEncoder != nil && (phase == "after-command" || phase == "end") {
			c := sys.chars[0][0]
			command := c.cmd[c.playerNo]
			var grantBuffer int32
			if grant, ok := command.Names["Grant"]; ok {
				for _, entry := range command.Commands[grant] {
					if entry.curbuftime > grantBuffer {
						grantBuffer = entry.curbuftime
					}
				}
			}
			if err := researchModeEncoder.Encode(map[string]interface{}{
				"case": researchCase, "introFrame": researchIntroFrame, "phase": phase,
				"tick": sys.tickCount, "roundState": sys.roundState(), "raw": researchBits(0),
				"var10": c.varGet(10).ToI(), "palNo": c.gi().palno, "grantBuffer": grantBuffer,
			}); err != nil {
				panic(err)
			}
		}
		return
	}
	if phase != "before-command" && phase != "after-command" && phase != "after-collision" && phase != "after-globalTick" && phase != "end" && phase != "hitdef" {
		return
	}

	chars := make([]map[string]interface{}, 0, 2)
	for pn := 0; pn < 2; pn++ {
		c := sys.chars[pn][0]
		commands := map[string][]int32{}
		cl := &c.cmd[c.playerNo]
		for _, name := range []string{"x", "y", "holdback", "holddown", "Grant"} {
			if idx, ok := cl.Names[name]; ok {
				for _, command := range cl.Commands[idx] {
					commands[name] = append(commands[name], command.curbuftime)
				}
			}
		}
		var frame interface{}
		if af := c.curFrame; af != nil {
			frame = map[string]interface{}{
				"sprite": [2]int32{af.Group, af.Number}, "duration": af.Time,
				"offset": [2]int32{af.Xoffset, af.Yoffset}, "clsn1": af.Clsn1, "clsn2": af.Clsn2,
			}
		}
		var helpers []int32
		for _, h := range sys.chars[pn][1:] {
			if h.helperId == 40000 {
				helpers = append(helpers, h.varGet(0).ToI())
			}
		}
		char := map[string]interface{}{
			"pn": pn + 1, "state": c.ss.no, "time": c.ss.time, "anim": c.animNo,
			"elem": c.anim.curelem, "elemTime": c.anim.curelemtime, "animTime": c.animTime(), "frame": frame,
			"bodyDistX": c.p2BodyDistX(c).ToF(), "palNo": c.gi().palno,
			"pos": researchNumbers(c.pos[:]...), "vel": researchNumbers(c.vel[:]...), "facing": c.facing, "ctrl": c.scf(SCF_ctrl),
			"raw": researchBits(pn), "commands": commands, "life": c.life,
			"hitPause": c.hitPauseTime, "hitShake": c.ghv.hitshaketime,
			"hitTime": c.ghv.hittime, "slideTime": c.ghv.slidetime, "ctrlTime": c.ghv.ctrltime,
			"hitVel": researchNumbers(c.ghv.xvel, c.ghv.yvel), "damage": c.ghv.damage,
			"guarded": c.ghv.guarded, "gethit": c.csf(CSF_gethit),
			"guardFlag": c.scf(SCF_guard), "guardPointDamage": c.ghv.guardpoints,
			"moveHit": c.moveHit(), "moveContact": c.moveContact(), "moveGuarded": c.moveGuarded(),
			"moveType": c.ss.moveType, "stateType": c.ss.stateType,
			"hitDef": map[string]interface{}{
				"attr": c.hitdef.attr, "guardFlag": c.hitdef.guardflag,
				"damage": c.hitdef.hitdamage, "guardDamage": c.hitdef.guarddamage,
				"hitPause": c.hitdef.pausetime, "guardPause": c.hitdef.guard_pausetime,
				"hitTime": c.hitdef.ground_hittime, "guardHitTime": c.hitdef.guard_hittime,
				"guardCtrlTime": c.hitdef.guard_ctrltime, "groundVel": researchNumbers(c.hitdef.ground_velocity[:]...),
				"guardVel": researchNumbers(c.hitdef.guard_velocity[:]...),
			},
			"vars": map[string]interface{}{
				"var6": c.varGet(6).ToI(), "var10": c.varGet(10).ToI(), "var12": c.varGet(12).ToI(), "var21": c.varGet(21).ToI(),
				"var40": c.varGet(40).ToI(), "fvar7": c.fvarGet(7).ToF(), "helper40000var0": helpers,
			},
			"friction": researchNumbers(c.getStandFriction(), c.getCrouchFriction()),
			"attackMul": c.attackMul[0], "attackBase": c.gi().attackBase, "finalDefense": c.finalDefense,
		}
		if researchStage16 || researchStage17 {
			char["vars"].(map[string]interface{})["var23"] = c.varGet(23).ToI()
		}
		if researchStage18 {
			helperX, helperY := researchCommandHelper(c)
			char["helperX"], char["helperY"] = helperX, helperY
			char["vars"].(map[string]interface{})["var2"] = c.varGet(2).ToI()
		}
		if researchWaveTrace {
			char["powerWaveCommand"] = c.commandByName(researchPowerWaveCommand)
			char["helperWave"] = researchPowerWaveHelper(c)
			char["vars"].(map[string]interface{})["var1"] = c.varGet(1).ToI()
			char["vars"].(map[string]interface{})["var2"] = c.varGet(2).ToI()
			char["vars"].(map[string]interface{})["fvar0"] = c.fvarGet(0).ToF()
			char["helpers"] = researchPowerWaveEntities(c)
			char["projectiles"] = len(sys.projs[pn])
			char["hitFall"] = c.ghv.fallflag
		}
		if researchStage20 || researchStage21 {
			char["vars"].(map[string]interface{})["var23"] = c.varGet(23).ToI()
			char["vars"].(map[string]interface{})["var31"] = c.varGet(31).ToI()
		}
		if researchStage21 {
			char["helperX"], char["helperY"] = researchCommandHelper(c)
			char["vars"].(map[string]interface{})["var28"] = c.varGet(28).ToI()
		}
		chars = append(chars, char)
	}
	row := map[string]interface{}{
		"case": researchCase, "frame": researchFrame, "phase": phase, "hitDefOwner": owner,
		"tick": sys.tickCount, "matchTime": sys.matchTime, "roundState": sys.roundState(),
		"chars": chars,
	}
	if researchStage17 || researchStage20 || researchStage21 {
		row["rng"] = sys.randseed
	}
	if researchWaveTrace {
		row["turbo"], row["slowTime"], row["globalPause"] = sys.turbo, sys.slowtime, sys.pausetime
	}
	if err := researchEncoder.Encode(row); err != nil {
		panic(err)
	}
}

func researchCommandHelper(c *Char) (int32, int32) {
	for _, helper := range sys.chars[c.playerNo][1:] {
		if helper.helperId == 99999 {
			return helper.varGet(0).ToI(), helper.varGet(2).ToI()
		}
	}
	return -1, -1
}

func researchStateChange(c *Char, next int32) {
	if !(researchStage18 || researchWaveTrace) || researchEncoder == nil || c.helperIndex != 0 {
		return
	}
	relevant := c.ss.no == 200 || c.ss.no == 205 || researchStage21 && c.ss.no == 215 || next == 200 || next == 205 || next == 210 || next == 215
	if researchWaveTrace {
		relevant = relevant || c.ss.no == 1000 || c.ss.no == 1050 || next == 1000 || next == 1050 || next == 300
	}
	if !relevant {
		return
	}
	helperX, helperY := researchCommandHelper(c)
	row := map[string]interface{}{
		"case": researchCase, "frame": researchFrame, "phase": "state-change", "tick": sys.tickCount,
		"seat": c.playerNo+1, "from": c.ss.no, "to": next, "stateTime": c.ss.time, "anim": c.animNo,
		"elem": c.anim.curelem, "elemTime": c.anim.curelemtime, "animTime": c.animTime(),
		"bodyDistX": c.p2BodyDistX(c).ToF(), "raw": researchBits(c.playerNo),
		"xBuffer": c.commandByName("x"), "yBuffer": c.commandByName("y"),
		"helperX": helperX, "helperY": helperY, "hitPause": c.hitPauseTime,
		"minus": c.minus, "ctrl": c.scf(SCF_ctrl), "var2": c.varGet(2).ToI(),
		"moveContact": c.moveContact(), "moveHit": c.moveHit(), "moveGuarded": c.moveGuarded(),
	}
	if researchWaveTrace {
		row["var1"] = c.varGet(1).ToI()
		row["powerWaveCommand"] = c.commandByName(researchPowerWaveCommand)
		row["helperWave"] = researchPowerWaveHelper(c)
	}
	if err := researchEncoder.Encode(row); err != nil {
		panic(err)
	}
}

func researchPowerWaveHelper(c *Char) int32 {
	for _, helper := range sys.chars[c.playerNo][1:] {
		if helper.helperId == 99999 {
			return helper.varGet(10).ToI()
		}
	}
	return -1
}

func researchPowerWaveEntities(c *Char) []map[string]interface{} {
	var result []map[string]interface{}
	for _, helper := range sys.chars[c.playerNo][1:] {
		if helper.helperId != 1000 && helper.helperId != 1050 {
			continue
		}
		var frame interface{}
		if af := helper.curFrame; af != nil {
			frame = map[string]interface{}{
				"sprite": [2]int32{af.Group, af.Number}, "duration": af.Time,
				"offset": [2]int32{af.Xoffset, af.Yoffset}, "clsn1": af.Clsn1, "clsn2": af.Clsn2,
			}
		}
		result = append(result, map[string]interface{}{
			"id": helper.id, "helperId": helper.helperId, "state": helper.ss.no, "anim": helper.animNo,
			"time": helper.ss.time, "facing": helper.facing, "pos": researchNumbers(helper.pos[:]...),
			"vel": researchNumbers(helper.vel[:]...), "moveContact": helper.moveContact(),
			"moveHit": helper.moveHit(), "moveGuarded": helper.moveGuarded(),
			"elem": helper.anim.curelem, "elemTime": helper.anim.curelemtime, "frame": frame,
			"hitPause": helper.hitPauseTime, "attackMul": helper.attackMul[0],
			"destroyed": helper.csf(CSF_destroy),
			"hitDef": map[string]interface{}{
				"attr": helper.hitdef.attr, "damage": helper.hitdef.hitdamage, "guardDamage": helper.hitdef.guarddamage,
				"hitPause": helper.hitdef.pausetime, "guardPause": helper.hitdef.guard_pausetime,
				"groundVel": researchNumbers(helper.hitdef.ground_velocity[:]...), "groundType": helper.hitdef.ground_type,
				"fall": helper.hitdef.ground_fall,
			},
		})
	}
	return result
}

func researchStage17Event(kind string, values map[string]interface{}) {
	if !(researchStage17 || researchStage21) || researchEncoder == nil {
		return
	}
	values["case"], values["frame"], values["tick"], values["phase"] = researchCase, researchFrame, sys.tickCount, kind
	if err := researchEncoder.Encode(values); err != nil {
		panic(err)
	}
}

func researchRandom(before, after int32) {

	}
}

func researchDamageVariable(c *Char, operation string, index, before, after int32) {
	if (researchStage17 || researchStage21) && researchEncoder != nil && c.helperIndex == 0 && (index == 6 || index == 23) {

		}

			"variable": index, "operation": operation, "before": before, "after": after, "hitPause": c.hitPauseTime,
		})
	}
}

func researchHitDef(c *Char) {
	if c.helperIndex == 0 && (c.ss.no == 200 || c.ss.no == 210 || c.ss.no == 215 || (researchStage16 || researchStage17 || researchStage18 || researchWaveTrace) && c.ss.no == 205) {
		researchTraceWithOwner("hitdef", c.playerNo+1)
	}
}

func researchEnd() {
	if !sys.tickNextFrame() {
		return
	}
	researchTrace("end")
	lastFrame := 75
	if researchWaveTrace {
		if configured, err := strconv.Atoi(os.Getenv("RESEARCH_LAST_FRAME")); err == nil && configured >= 75 {
			lastFrame = configured
		}
	}
	if researchFrame >= lastFrame {
		if err := researchTraceFile.Close(); err != nil {
			panic(err)
		}
		fmt.Fprintln(os.Stderr, "COMBAT_RESEARCH_DONE", researchCase)
		os.Exit(0)
	}
}
